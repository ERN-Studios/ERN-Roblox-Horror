263: 		return {MinX=left.X + left.W * .5 - .02, MaxX=right.X - right.W * .5 + .02,
264: 			MinZ=a.Z - halfWidth, MaxZ=a.Z + halfWidth}
265: 	end
266: 	local north, south = if a.Z < b.Z then a else b, if a.Z < b.Z then b else a
267: 	return {MinX=a.X - halfWidth, MaxX=a.X + halfWidth,
268: 		MinZ=north.Z + north.D * .5 - .02, MaxZ=south.Z - south.D * .5 + .02}
269: end

271: local function makeHash(layout)
272: 	local pieces = {
273: 		tostring(VERSION),
274: 		tostring(layout.Seed),
275: 		tostring(layout.Attempt),
276: 	}
277: 	for _, room in ipairs(layout.Rooms) do
278: 		table.insert(pieces, string.format(
279: 			"R:%s:%.3f:%.3f:%d:%d:%d:%s:%s:%s:%d:%d:%d",
280: 			room.Id, room.X, room.Z, room.W, room.D, room.H,
281: 			room.Kind, room.Decor, room.ThemeId, room.SectionIndex,
282: 			if room.Module then 1 else 0, room.LocalSeed
283: 		))
284: 	end
285: 	for _, link in ipairs(layout.Links) do
286: 		table.insert(pieces, string.format(
287: 			"L:%s:%s:%s:%s:%s:%d:%d",
288: 			link.Id, link.A, link.B, link.Door, link.ThemeId,
289: 			link.SectionIndex, if link.Gateway then 1 else 0
290: 		))
291: 	end
292: 	local text = table.concat(pieces, ";")
293: 	local hash = 5381
294: 	for index = 1, #text do
295: 		hash = (hash * 33 + string.byte(text, index)) % 4294967296
296: 	end
297: 	return string.format("L3-%d-%08x", VERSION, hash)
298: end

733: 	layout.Roles = {

760: function LayoutGenerator.Validate(layout)
761: 	if type(layout) ~= "table" then return fail("layout is not a table") end
762: 	if layout.Version ~= VERSION then return fail("layout version is stale") end
763: 	if type(layout.Rooms) ~= "table" or #layout.Rooms ~= DISTRICT_COUNT * ROOMS_PER_DISTRICT + 2 then
764: 		return fail("layout must contain exactly 26 rooms")
765: 	end
766: 	local expectedLinks = DISTRICT_COUNT
767: 		* ((ROOMS_PER_DISTRICT - 1) + Tuning.ExtraLinksPerDistrict)
768: 		+ ENTRY_AND_GATEWAY_LINKS
769: 	if type(layout.Links) ~= "table" or #layout.Links ~= expectedLinks then
770: 		return fail(string.format("layout must contain exactly %d links", expectedLinks))
771: 	end
772: 	if type(layout.Districts) ~= "table" or #layout.Districts ~= DISTRICT_COUNT then
773: 		return fail("layout must contain exactly three districts")
774: 	end
775: 	if type(layout.RoomById) ~= "table" or type(layout.Adjacency) ~= "table"
776: 		or type(layout.GatewayLinks) ~= "table" then
777: 		return fail("layout lookup tables are missing")
778: 	end
