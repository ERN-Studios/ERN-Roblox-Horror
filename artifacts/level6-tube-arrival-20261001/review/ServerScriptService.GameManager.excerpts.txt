713 local function freeElevatorFrame(player, pad)
714  local levelOne = pad:GetAttribute("Level2_CompatibilityMarker") ~= true
715   and pad:GetAttribute("Level3_CompatibilityMarker") ~= true
716  local forward = levelOne and Vector3.xAxis or pad.CFrame.LookVector
717  local side = levelOne and Vector3.zAxis or pad.CFrame.RightVector
718  local depth = pad.Size.X + 3 -- Level 1's emergency pad is 3 studs shorter than its floor.
719  local rows = levelOne and math.max(1, math.floor((depth - 6) / 4) + 1) or 7
720  for row = 0, rows - 1 do
721   for column = 0, (levelOne and 1 or 3) do
722    local across = (column % 2 == 0 and -1 or 1) * (2 + math.floor(column / 2) * 4)
723    local along = levelOne and (depth / 2 - 3 - row * 4) or row * 4
724    local position = pad.Position + side * across + forward * along + Vector3.new(0, 4, 0)
725    if arrivalPointFree(player, position) then
726     return CFrame.lookAt(position, position + forward)
727    end
728   end
729  end
730  return nil

874     for slot = 0, MAX_PLAYERS_PER_STATION - 1 do
875      -- Keep each successive actual-root placement farther UP the same bore.
876      -- The rear reserve prevents any candidate from crossing its physical cap.
877      local alpha = startAlpha + slot * 12 / length
878      if alpha <= .025 or (1 - alpha) * length < 12 then break end

882       return {Position = point, Tangent = direction, Velocity = direction * velocity.Magnitude}
883      end
