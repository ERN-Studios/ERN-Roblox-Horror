30 local function ownedFolder(parent: Instance, name: string): Folder
31     local old = parent:FindFirstChild(name)
32     if old then assert(old:IsA("Folder"), name .. " has the wrong class") return old end
33     local object = Instance.new("Folder")
34     object.Name = name
35     object.Parent = parent
36     return object
37 end

43 local function infrastructure()
44     local remotes = ownedFolder(ReplicatedStorage, Configuration.RemotesFolderName)
45     for _, spec in ipairs({
46         {Configuration.ClientEventName, "RemoteEvent"},
47         {Configuration.HideRequestEventName, "RemoteEvent"},
48         {Configuration.MallManagerMotionEventName, "UnreliableRemoteEvent"},
49     }) do
50         local old = remotes:FindFirstChild(spec[1])
51         if old then
52             assert(old.ClassName == spec[2], "Level 6 remote class mismatch")
53         else
54             local object = Instance.new(spec[2])
55             object.Name = spec[1]
56             object.Parent = remotes
57         end
58     end
59 end

190 local function bindManager(manifest: any)
191     disconnectManager()
192     local token = managerToken
193     local function sync()
194         if token ~= managerToken or activeManifest ~= manifest
195             or not manifest.World or manifest.World.Parent ~= workspace then return end
196         if workspace:GetAttribute("Level6MallManagerHuntActive") ~= true
197             or workspace:GetAttribute("Level6RoundActive") ~= true or countParticipants() == 0 then
198             MallManagerController.Stop()
199             return
200         end
201         if MallManagerController.GetSnapshot() then return end
202         local ok, result = pcall(MallManagerController.Start, manifest, generation)
203         if not ok then
204             state():SetAttribute("Level6_ManagerError", tostring(result))
205             warn("[Level 6 Preview] Manager start failed:", result)
206             return
207         end
208         if result == nil then
209             task.delay(.35, function()
210                 if token == managerToken then sync() end
211             end)
212         end
213     end
214     managerConnection = workspace:GetAttributeChangedSignal("Level6MallManagerHuntActive"):Connect(sync)
215     sync()
216 end

239 function Runtime.Leave(player: Player)
240     local record = participants[player]
241     if not record then
242         if player:GetAttribute("Level6InRound") == true then player:SetAttribute("Level6InRound", nil) end
243         return
244     end
245     -- Drop while the participant gate is still true; another dev can recover their CDs.
246     pcall(ObjectiveController.ReleaseParticipant, player)
247     pcall(HidingController.ReleaseParticipant, player)
248     participants[player] = nil
249     for _, connection in ipairs(record.Connections) do connection:Disconnect() end
250     player:SetAttribute("Level6InRound", nil)
251     player:SetAttribute("Level6Escaped", nil)
252     player:SetAttribute("Level6BeingChased", nil)
253     player:SetAttribute("Level6PreviewActive", nil)
254     if countParticipants() == 0 then
255         stopControllers()
256         destroyOwnedWorlds()
257         activeManifest = nil
258         stale = false
259         workspace:SetAttribute("Level6SelectedLevel", nil)
260         applyBaselineReplicatedState(state(), nil)
261     end
262 end

264 local function returnToLobby(player: Player)
265     local character = player.Character
266     local root = character and character:FindFirstChild("HumanoidRootPart")
267     local spawn = lobbySpawn()
268     if root and root:IsA("BasePart") and spawn then
269         pcall(function() player:RequestStreamAroundAsync(spawn.Position, 8) end)
270         -- A new character or round supersedes the old preview return.
271         if player.Character == character and player:GetAttribute("InRound") ~= true then
272             Runtime.Leave(player)
273             root.AssemblyLinearVelocity = Vector3.zero
274             root.AssemblyAngularVelocity = Vector3.zero
275             character:PivotTo(spawn.CFrame * CFrame.new(0, 4, 0))
276             return
277         end
278     end
279     Runtime.Leave(player)
280 end

282 function Runtime.Cleanup()

294 function Runtime.EnsureWorld(): (Model, BasePart)
295     assert(RunService:IsRunning(), "Level 6 preview construction runs during Play")
296     if building then
297         -- The revised 279-chunk bake can outlast the old four-minute wait;
298         -- concurrent preview joins must wait for the same owned build.
299         local deadline = os.clock() + 1260
300         repeat task.wait(.05) until not building or os.clock() > deadline
301         assert(not building, "Level 6 build is still busy")
302         assert(not buildError, buildError)
303     end
304     if activeManifest and activeManifest.World.Parent == workspace and not stale then
305         return activeManifest.World, activeManifest.PreviewExit
306     end
307     assert(countParticipants() == 0, "Cannot regenerate beneath a preview participant")
308     building = true
309     buildError = nil
310     local ok, result = xpcall(function()
311         Runtime.Cleanup()
312         generation += 1
313         infrastructure()
314         local requested = workspace:GetAttribute("Level6Seed")
315         local pinned = type(requested) == "number" and requested == requested
316             and requested >= 1 and requested < MAX_SEED
317         local seed = if pinned then math.floor(requested) else Random.new():NextInteger(1, MAX_SEED - 1)
318         applyBaselineReplicatedState(state(), {
319             Phase="GENERATING_LAYOUT", RequestedSeed=seed, SeedPinned=pinned,
320             Generation=generation, ModuleGoal=Configuration.ModuleGoal,
321             LightingMode="NORMAL", RoomSongPhase="WAITING_FOR_ROUND",
322         })
323         workspace:SetAttribute("Level6SelectedLevel", 6)
324         workspace:SetAttribute("Level6RoundActive", false)
325         local layout = LayoutGenerator.Generate(seed)
326         local valid, problem = LayoutGenerator.Validate(layout)
327         assert(valid, problem)
328         local s = state()
329         s:SetAttribute("Level6_Phase", "BAKING_BLENDER_KIT")
330         local kit = RuntimeBake.Ensure()
331         assert(kit and kit:GetAttribute("Ready") == true, "Level 6 Blender kit is not ready")
332         s:SetAttribute("Level6_KitBakedCount", kit:GetAttribute("BakedCount") or 0)
333         s:SetAttribute("Level6_KitBakeSeconds", kit:GetAttribute("BakeSeconds") or 0)
334         s:SetAttribute("Level6_Phase", "BUILDING_WORLD")
335         local began = os.clock()
336         local manifest = WorldBuilder.Build(layout, generation)
337         activeManifest = manifest
338         manifest.Participants = participants
339         -- Root-owned adapter replaces every visible shell/furniture piece with the imported Blender kit.
340         VisualAdapter.Apply(manifest)
341         local marker = Instance.new("Part")
342         marker.Name = "Level6Exit"
343         marker.Size = Vector3.new(2, .2, 2)
344         marker.CFrame = manifest.ElevatorSpawn.CFrame * CFrame.new(0, 3.5, 0)
345         marker.Transparency = 1
346         marker.Anchored = true
347         marker.CanCollide = false
348         marker.CanQuery = false
349         marker.CanTouch = false
350         marker:SetAttribute("Level6PreviewReturn", true)
351         marker.Parent = manifest.World
352         manifest.PreviewExit = marker
353         for _, key in ipairs({"Level6Preview", "PreviewOnly", "Level6PreviewReady"}) do
354             manifest.World:SetAttribute(key, true)
355         end
356         manifest.World:SetAttribute("Level6BuildOwned", true)
357         manifest.World:SetAttribute("Level6_RoomFloorArea", layout.RoomFloorArea)
358         manifest.World:SetAttribute("Level6_AuthoredGatewayFloorArea", layout.BlenderRoomFloorArea)
359         s:SetAttribute("Level6_BuildSeconds", os.clock() - began)
360         s:SetAttribute("Level6_WorldDescendants", #manifest.World:GetDescendants())
361         s:SetAttribute("Level6_ResolvedSeed", layout.ResolvedSeed)
362         s:SetAttribute("Level6_LayoutHash", layout.LayoutHash)
363         s:SetAttribute("Level6_GeneratedRoomCount", #layout.Rooms)
364         s:SetAttribute("Level6_AuthoredGatewayRoomCount", #layout.BlenderRooms)
365         s:SetAttribute("Level6_AuthoredGatewayFloorArea", layout.BlenderRoomFloorArea)
366         s:SetAttribute("Level6_GeneratedCorridorCount", #layout.Links)
367         s:SetAttribute("Level6_GeneratedDistrictCount", #layout.Districts)
368         s:SetAttribute("Level6_ExitPosition", manifest.ExitPosition)
369         s:SetAttribute("Level6_BlackoutScreamOpeningCount", #manifest.BlackoutScreamOpenings)
370         s:SetAttribute("Level6_Phase", "READY")
371         return manifest
372     end, debug.traceback)
373     building = false
374     if not ok then
375         buildError = tostring(result)
376         stopControllers()
377         destroyOwnedWorlds()
378         activeManifest = nil
379         workspace:SetAttribute("Level6SelectedLevel", nil)
380         state():SetAttribute("Level6_Error", buildError)
381         error(buildError)
382     end
383     return result.World, result.PreviewExit
384 end

386 function Runtime.Join(player: Player): (boolean, string?)
387     if player.Parent ~= Players or not DevAccess.IsLevel6PreviewAllowed(player)
388         or player:GetAttribute("InRound") == true
389         or workspace:GetAttribute("ReservedRoundServer") == true then
390         return false, "PREVIEW_NOT_ALLOWED"
391     end
392     if participants[player] then return true end
393     local manifest = activeManifest
394     local character = player.Character
395     local humanoid = character and character:FindFirstChildOfClass("Humanoid")
396     local root = humanoid and humanoid.RootPart
397     if not manifest or stale or not root or root.Anchored or humanoid.Health <= 0
398         or (root.Position - manifest.PreviewExit.Position).Magnitude > 18 then
399         return false, "ARRIVAL_NOT_READY"
400     end
401     local record = {Connections={}}
402     participants[player] = record
403     player:SetAttribute("Level6Escaped", false)
404     player:SetAttribute("Level6InRound", true)
405     player:SetAttribute("Level6PreviewActive", true)
406     table.insert(record.Connections, player:GetAttributeChangedSignal("Level6Escaped"):Connect(function()
407         if player:GetAttribute("Level6Escaped") == true then task.defer(returnToLobby, player) end
408     end))
409     table.insert(record.Connections, player:GetAttributeChangedSignal("InRound"):Connect(function()
410         if player:GetAttribute("InRound") == true then Runtime.Leave(player) end
411     end))
412     table.insert(record.Connections, humanoid.Died:Connect(function()
413         task.defer(Runtime.Leave, player)
414     end))
415     table.insert(record.Connections, player.CharacterRemoving:Connect(function()
416         Runtime.Leave(player)
417     end))
418     if not playing then
419         local ok, problem = xpcall(function()
420             ObjectiveController.Start(manifest, generation)
421             HidingController.Start(manifest, generation)
422             MusicSequenceController.Start(manifest, generation)
423             workspace:SetAttribute("Level6LightingOwnedByController", true)
424             workspace:SetAttribute("Level6RoundActive", true)
425             playing = true
426             bindManager(manifest)
427         end, debug.traceback)
428         if not ok then
429             Runtime.Leave(player)
430             state():SetAttribute("Level6_Error", tostring(problem))
431             return false, tostring(problem)
432         end
433     end
434     return true
435 end
