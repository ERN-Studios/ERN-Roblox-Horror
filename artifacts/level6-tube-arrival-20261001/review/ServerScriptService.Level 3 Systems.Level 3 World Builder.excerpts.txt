1845 	local riders = setmetatable({}, {__mode = "k"})
1846 	local slipAccumulator = 0
1847 	local slipConnection: RBXScriptConnection?
1848 	slipConnection = RunService.Heartbeat:Connect(function(deltaTime)
1849 		if not visual:IsDescendantOf(workspace) then
1850 			if slipConnection then slipConnection:Disconnect() end
1851 			-- The world can be torn down with a rider still in the bore. Release
1852 			-- every PlatformStand this loop set, or that player stays a ragdoll
1853 			-- for the rest of their session.
1854 			for _, player in ipairs(Players:GetPlayers()) do
1855 				local character = player.Character
1856 				local humanoid = character and character:FindFirstChildOfClass("Humanoid")
1857 				if humanoid and character:GetAttribute("Level3_ProgressionSliding") == true then
1858 					humanoid.PlatformStand = false
1859 					character:SetAttribute("Level3_ProgressionSliding", nil)
1860 				end
1861 			end
1862 			return
1863 		end
1864 		slipAccumulator += deltaTime
1865 		if slipAccumulator < .08 then return end
1866 		slipAccumulator = 0
1867 		for _, player in ipairs(Players:GetPlayers()) do
1868 			local character = player.Character
1869 			local humanoid = character and character:FindFirstChildOfClass("Humanoid")
1870 			local root = character and character:FindFirstChild("HumanoidRootPart")
1871 			if humanoid and humanoid.Health > 0 and root and root:IsA("BasePart") then
1872 				if player:GetAttribute("InRound") == true and workspace:GetAttribute("SelectedLevel") == 3 then
1873 					riders[character] = SlideSafety.Apply(visual, character, riders[character], true) or nil
1874 				end
1875 				local alpha = (mouthX - root.Position.X) / tubeLength
1876 				local pathY = centerY + rise * alpha * alpha
1877 				local insideBore = alpha > .025 and alpha < .99
1878 					and math.abs(root.Position.Z - p.Z) < radius - .8
1879 					and math.abs(root.Position.Y - pathY) < radius + 2
1880 				if insideBore then
1881 					-- LEVEL2_EXIT_TRANSITION_20260828
1882 					-- This is the tail of Level 2's flume, so it has to ride like
1883 					-- one. PlatformStand takes the humanoid controller out of the
1884 					-- way: without it the character stays in Running, resists the
1885 					-- imposed velocity and WALKS the bore at about 16 studs/s,
1886 					-- which reads as a corridor rather than the end of a slide.
1887 					if not humanoid.PlatformStand then
1888 						humanoid.PlatformStand = true
1889 						character:SetAttribute("Level3_ProgressionSliding", true)
1890 					end
1891 					-- Push along the live path tangent rather than world +X, so
1892 					-- the steep rear section drives the rider forward instead of
1893 					-- straight down into the shell.
1894 					local tangent = Vector3.new(tubeLength, -2 * rise * alpha, 0).Unit
1895 					-- The shared slide controller owns production client physics and
1896 					-- releases the actual body joints. Keep this velocity write only as
1897 					-- a server fallback until its replicated ragdoll session is live.
1898 					if character:GetAttribute("Level2_RagdollServerActive") ~= true then
1899 						local velocity = root.AssemblyLinearVelocity
1900 						local along = velocity:Dot(tangent)
1901 						local lateral = velocity - tangent * along
1902 						root.AssemblyLinearVelocity = tangent * math.max(along, SLIDE_MINIMUM_SPEED)
1903 							+ lateral * .35
1904 					end
1905 				elseif character:GetAttribute("Level3_ProgressionSliding") == true then
1906 					-- Out of the bore: hand control straight back. Leaving
1907 					-- PlatformStand set would drop the arriving player onto the
1908 					-- mall floor as an inert ragdoll.
1909 					-- If the shared ragdoll service is still live, let its client End
1910 					-- restore joints and humanoid state atomically instead of racing it.
1911 					if character:GetAttribute("Level2_RagdollServerActive") ~= true then
1912 						humanoid.PlatformStand = false
1913 						character:SetAttribute("Level3_ProgressionSliding", nil)
1914 					end
1915 				end
1916 			end
1917 		end
1918 	end)
