def save(path, raw):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("xb") as stream:
        assert stream.write(raw) == len(raw)
    assert path.read_bytes() == raw


def encoded(value):
    return (json.dumps(value, indent=2, ensure_ascii=False) + "\n").encode()


def lua_literal(text):
    level = "="
    while "]" + level + "]" in text:
        level += "="
    return "string.sub([" + level + "[_" + text + "]" + level + "],2)"

    assert entry["sha256"] == BEFORE and entry["sourceEditorMatches"] is True

            else:

    pinned = {

def execute(phase,receipt):
    assert receipt is not None and not receipt.exists(), "Require a new immutable native receipt"
    proof=json.loads((DEST/"proposal.json").read_bytes())
    code=(DEST/("native-"+phase+".luau")).read_bytes()
    assert sha(code)==proof["nativeCodeHashes"][phase]
    assert sha(lf((ROOT/FILE).read_bytes()))==proof["beforeSHA256" if phase=="before" else "candidateSHA256"]
    sys.path.insert(0,str(ROOT/"tools"))
    from sync_from_studio import StudioMcpClient,find_mcp_batch,select_studio,RELEASE_STUDIO_NAME,require_edit_state
    client=StudioMcpClient(find_mcp_batch());result=None;error=None
    try:
        client.initialize();studio=select_studio(client,RELEASE_STUDIO_NAME,15)
        assert studio["id"]=="1b603e69-7ae1-4572-b210-d5bf05c684ff"
        require_edit_state(client,studio["id"])
        result=json.loads(client.call("execute_luau",{"studio_id":studio["id"],"datamodel_type":"Edit","code":code.decode()}))
    except BaseException as primary:error=primary
    finally:
        try:client.close()
        except BaseException as cleanup:
            if error is None:error=cleanup
            else:error.add_note("MCP close: "+str(cleanup))
        if result is None:result={"pass":False}
        if error is not None:result.update({"pass":False,"transportError":str(error)})
        result["nativeInputSHA256"]=sha(code);save(receipt,encoded(result))
    print(json.dumps({k:result.get(k) for k in ("pass","contractPassed","originalsUnchanged","probesCreated","probesDestroyed","remainingProbes","error","transportError")}))
    assert error is None and result.get("pass") is True,"Native gate failed; retain the candidate"


if __name__=="__main__":
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--execute",choices=("before","installed"));parser.add_argument("--receipt",type=Path)
    args=parser.parse_args()
    if args.execute:execute(args.execute,args.receipt)
    else:prepare()
